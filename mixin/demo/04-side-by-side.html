1. Mã nguồn đối chiếu

Cùng một ý đồ. Chú ý những dòng được đánh dấu — đó là toàn bộ khác biệt.

Polymer 3

const OpenableMixin = dedupingMixin((sc) =>
  class extends sc {
    static get properties() {
      return {
        opened: {
          type: Boolean,
          value: false,
          reflectToAttribute: true,
          observer: 'openedChanged_',
        },
        nhan: {computed: 'tinhNhan_(opened)'},
      };
    }

    toggle() { this.opened = !this.opened; }
    open()   { this.opened = true; }
    close()  { this.opened = false; }

    tinhNhan_(o) { return o ? 'Đang mở' : 'Đang đóng'; }

    openedChanged_(moi) {
      this.dispatchEvent(new CustomEvent(
          'opened-changed', {detail: {value: moi}}));
    }
  });

Lit 3

const OpenableMixin = dedupeMixin((Base) =>
  class extends Base {
    static properties = {
      opened: {
        type: Boolean,
        reflect: true,
      },
    };

    constructor() {
      super();
      this.opened = false;
    }

    get nhan() { return this.opened ? 'Đang mở' : 'Đang đóng'; }

    toggle() { this.opened = !this.opened; }
    open()   { this.opened = true; }
    close()  { this.opened = false; }

    updated(changed) {
      super.updated(changed);
      if (changed.has('opened')) {
        this.dispatchEvent(new CustomEvent(
            'opened-changed', {detail: {value: this.opened}}));
      }
    }
  });
toggle(), open(), close() giống hệt nhau từng chữ. Chỉ khác ở vỏ khai báo: deduping, cú pháp properties, chỗ đặt giá trị mặc định, tên reflect, và cách phản ứng khi property đổi (observer vs updated()).
